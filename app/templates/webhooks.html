İmzayı doğrulama (Python) 
import hashlib, hmac

def dogru_mu(govde: bytes, imza_basligi: str, anahtar: str) -> bool:
    beklenen = "sha256=" + hmac.new(anahtar.encode(), govde, hashlib.sha256).hexdigest()
    return hmac.compare_digest(beklenen, imza_basligi)
